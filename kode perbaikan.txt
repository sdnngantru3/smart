var REQUEST_SPREADSHEET_ID = '';

function getRequestSpreadsheet_() {
  return REQUEST_SPREADSHEET_ID ? SpreadsheetApp.openById(REQUEST_SPREADSHEET_ID) : SpreadsheetApp.getActiveSpreadsheet();
}

function hasHtmlFile_(name) {
  try { DriveApp.getFilesByName(name + '.html'); return true; } catch (e) { return false; }
}

function doGet(e) {
  e = e || { parameter: {} };
  var params = e.parameter || {};
  var action = params.action || '';

  // API-first: GitHub Pages menggunakan endpoint ini. Tidak lagi bergantung pada index.html di Apps Script.
  if (action) {
    REQUEST_SPREADSHEET_ID = String(params.spreadsheetId || params.SPREADSHEET_ID || '').trim();
    return handleRequest(action, params, params);
  }

  // Backward compatibility: bila file kuis masih berada di Apps Script, tetap bisa dibuka.
  if (params.page === 'kuis') {
    try {
      return HtmlService.createHtmlOutputFromFile('kuis')
        .setTitle('Kuis Interaktif - EduSmart LMS')
        .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
    } catch (err) {
      return jsonOutput_({success:false, message:'File kuis tidak ada di project Apps Script. Gunakan kuis.html dari GitHub Pages.', data:null});
    }
  }

  return jsonOutput_({
    success: true,
    message: 'EduSmart LMS API aktif. Gunakan parameter action untuk memanggil API.',
    data: { service: 'EduSmart LMS Google Apps Script API', version: 'GitHub Pages API v1' }
  });
}

function doPost(e) {
  e = e || { parameter: {}, postData: { contents: '' } };
  var params = e.parameter || {};
  var body = {};
  var raw = e.postData && e.postData.contents ? String(e.postData.contents) : '';
  if (raw) {
    try { body = JSON.parse(raw); } catch (err) {
      // GitHub Pages mengirim application/x-www-form-urlencoded agar request CORS tetap sederhana.
      body = {};
    }
  }
  // Form-urlencoded juga tersedia melalui e.parameter. Salin field JSON yang dikirim frontend.
  Object.keys(params).forEach(function(k) {
    if (body[k] === undefined) body[k] = params[k];
  });
  // Field kompleks dari GitHub Pages dikirim sebagai string JSON.
  ['records', 'result', 'intent'].forEach(function(k) {
    if (typeof body[k] === 'string') {
      var rawValue = String(body[k]).trim();
      if (rawValue && (rawValue[0] === '[' || rawValue[0] === '{')) {
        try { body[k] = JSON.parse(rawValue); } catch (ignore) {}
      }
    }
  });
  var action = params.action || body.action || '';
  REQUEST_SPREADSHEET_ID = String(body.spreadsheetId || body.SPREADSHEET_ID || params.spreadsheetId || params.SPREADSHEET_ID || '').trim();
  return handleRequest(action, params, body);
}

function handleRequest(action, params, body) {
  action = String(action || (body && body.action) || (params && params.action) || '').trim();
  params = params || {};
  body = body || {};
  if (!REQUEST_SPREADSHEET_ID) {
    REQUEST_SPREADSHEET_ID = String(body.spreadsheetId || body.SPREADSHEET_ID || params.spreadsheetId || params.SPREADSHEET_ID || '').trim();
  }

  var result = { success: false, message: 'Action tidak dikenal: ' + action, data: null };
  try {
    var ss = getRequestSpreadsheet_();
    if (action === 'health') {
      result = { success: true, message: 'EduSmart LMS API aktif.', data: { service: 'Google Apps Script API', spreadsheetId: ss.getId(), title: ss.getName() } };
    }
    else if (action === 'testConnection') {
      result = { success: true, message: 'Koneksi ke Google Spreadsheet Berhasil!', data: { title: ss.getName(), spreadsheetId: ss.getId() } };
    }
    else if (action === 'setupDatabase') result = setupDatabase();
    else if (action === 'saveAppConfig') result = saveAppConfigToSheet(body || {});
    else if (action === 'getAppConfig') result = { success: true, message: 'OK', data: getAppConfigFromSheet() };
    else if (action === 'login') result = body && body.spreadsheetId ? loginUserBySpreadsheetId(body.spreadsheetId, body.username, body.password, body.role) : loginUser(body.username, body.password, body.role);
    else if (action === 'getGuruProfile') result = getGuruProfile(body || {});
    else if (action === 'saveGuruProfile') result = saveGuruProfile(body || {});
    else if (action === 'getMenuGuru') result = getMenuGuru(body || {});
    else if (action === 'saveMenuGuru') result = saveMenuGuru(body || {});
    else if (action === 'deleteMenuGuru') result = deleteMenuGuru(body || {});
    else if (action === 'reorderMenuGuru') result = reorderMenuGuru(body || {});
    else if (action === 'getStudents') result = getStudentsByClass((body && body.kelas) || params.kelas || '');
    else if (action === 'saveStudent') result = saveStudentAndCreateUser(body);
    else if (action === 'importStudents') result = importStudentsAndCreateUsers(body.records || []);
    else if (action === 'deleteStudent') result = deleteStudentAndUser(body.key);
    else if (action === 'getSubjects') result = getSheetData('MasterMapel');
    else if (action === 'saveSubject') result = addOrUpdateRow('MasterMapel', body, 'Kode');
    else if (action === 'importSubjects') result = importRowsToSheet('MasterMapel', body.records || [], 'Kode');
    else if (action === 'getAttendance') result = getSheetData('Absensi');
    else if (action === 'saveAttendance') result = appendMultipleRows('Absensi', body.records);
    else if (action === 'importAttendance') result = importRowsToSheet('Absensi', body.records || [], 'ID');
    else if (action === 'getGrades') result = getSheetData('Nilai');
    else if (action === 'saveGrade') result = saveGradeAndRebuildPenilaian(body || {});
    else if (action === 'saveGradesBatch') result = saveGradesBatchAndRebuildPenilaian(body || {});
    else if (action === 'importGrades') result = importGradesAndRebuild(body.records || []);
    else if (action === 'getPenilaianConfig') result = getPenilaianConfig();
    else if (action === 'savePenilaianConfig') result = savePenilaianConfig(body || {});
    else if (action === 'deletePenilaianConfig') result = deletePenilaianConfig(body || {});
    else if (action === 'rebuildPenilaianSheet') result = rebuildPenilaianSheet();
    else if (action === 'getPenilaianSheet') result = getSheetData('Penilaian');
    else if (action === 'getJournals') result = getSheetData('Jurnal');
    else if (action === 'saveJournal') result = addOrUpdateRow('Jurnal', body, 'ID');
    else if (action === 'importJournals') result = importRowsToSheet('Jurnal', body.records || [], 'ID');
    else if (action === 'getQuizzes') result = getSheetData('Kuis');
    else if (action === 'getQuizQuestions') result = getSheetData('SoalKuis');
    else if (action === 'saveQuizBundle') result = saveQuizBundle(body.spreadsheetId || REQUEST_SPREADSHEET_ID, body.quiz, body.questions);
    else if (action === 'saveQuizResult') result = addOrUpdateRow('HasilKuis', body, 'ID');
    else if (action === 'getQuizResults') result = getSheetData('HasilKuis');
    else if (action === 'getAnnouncements') result = getSheetData('Pengumuman');
    else if (action === 'saveAnnouncement') result = addOrUpdateRow('Pengumuman', body, 'ID');
    else if (action === 'importAnnouncements') result = importRowsToSheet('Pengumuman', body.records || [], 'ID');
    else if (action === 'getOpenAiStatus') result = getOpenAiStatus(body || {});
    else if (action === 'saveOpenAiApiKey') result = saveOpenAiApiKey(body || {});
    else if (action === 'haloHalo') result = haloHalo(body || {});
    else if (action === 'haloHaloConfirm') result = haloHaloConfirm(body || {});
    else if (action === 'syncOfflineData') result = syncOfflineData(body || {});
    else if (action === 'resetDatabase') result = clearAllData();
    else if (action === 'quizGetConfig') result = quizGetConfig(body.spreadsheetId || REQUEST_SPREADSHEET_ID);
    else if (action === 'quizLogin') result = quizLogin(body.spreadsheetId || REQUEST_SPREADSHEET_ID, body.username, body.password);
    else if (action === 'quizLoginByNis') result = quizLoginByNis(body.spreadsheetId || REQUEST_SPREADSHEET_ID, body.nis, body.kelas);
    else if (action === 'getInteractiveQuizData') result = getInteractiveQuizData(body.spreadsheetId || REQUEST_SPREADSHEET_ID, body.nis, body.kelas);
    else if (action === 'saveInteractiveQuizResult') result = saveInteractiveQuizResult(body.spreadsheetId || REQUEST_SPREADSHEET_ID, body.result);
  } catch(err) {
    result = { success: false, message: err && err.message ? err.message : String(err), data: null };
  }
  return jsonOutput_(result);
}

function jsonOutput_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj || {})).setMimeType(ContentService.MimeType.JSON);
}

function ensureDatabaseSheets() {
  var ss = getRequestSpreadsheet_();
  var schemas = getDatabaseSchemas();
  var created = [];
  var repaired = [];
  var sheetNames = Object.keys(schemas);

  sheetNames.forEach(function(sheetName) {
    var headers = schemas[sheetName];
    var sheet = ss.getSheetByName(sheetName);
    if (!sheet) {
      sheet = ss.insertSheet(sheetName);
      sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
      created.push(sheetName);
    } else if (sheet.getLastRow() === 0) {
      sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
      repaired.push(sheetName);
    } else {
      var width = Math.max(sheet.getLastColumn(), headers.length);
      var current = sheet.getRange(1, 1, 1, width).getValues()[0];
      var mismatch = false;
      for (var i = 0; i < headers.length; i++) {
        if (String(current[i] == null ? '' : current[i]).trim() !== headers[i]) {
          mismatch = true;
          break;
        }
      }
      if (mismatch && sheet.getLastRow() <= 1) {
        sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
        repaired.push(sheetName);
      }
    }
    sheet.getRange(1, 1, 1, headers.length).setFontWeight('bold').setBackground('#e2e8f0');
    if (sheet.getFrozenRows() < 1) sheet.setFrozenRows(1);
  });

  return { created: created, repaired: repaired, total: sheetNames.length, names: sheetNames };
}

function saveAppConfigToSheet(input) {
  var ss = getRequestSpreadsheet_();
  ensureDatabaseSheets();
  var sheet = ss.getSheetByName('Config');
  if (!sheet) {
    sheet = ss.insertSheet('Config');
    sheet.appendRow(['Key', 'Value', 'Description', 'UpdatedAt']);
    sheet.getRange(1, 1, 1, 4).setFontWeight('bold').setBackground('#e2e8f0');
  }

  input = input || {};
  var existingConfig = getAppConfigFromSheet();
  var spreadsheetId = String(input.SPREADSHEET_ID || existingConfig.SPREADSHEET_ID || ss.getId()).trim();
  var scriptUrl = String(input.APPSCRIPT_URL || input.WEB_APP_URL || existingConfig.APPSCRIPT_URL || '').trim();
  var scriptId = String(existingConfig.SCRIPT_ID || '').trim();
  try { scriptId = ScriptApp.getScriptId() || scriptId; } catch (e) {}
  var now = new Date().toISOString();

  var configRows = [
    ['SPREADSHEET_ID', spreadsheetId, 'ID Spreadsheet database utama', now],
    ['APPSCRIPT_URL', scriptUrl, 'URL Web App Google Apps Script (/exec)', now],
    ['SCRIPT_ID', scriptId, 'ID project Google Apps Script', now],
    ['DEFAULT_STUDENT_PASSWORD', '123456', 'Password default akun siswa otomatis', now],
    ['GURU_USERNAME', String(input.GURU_USERNAME || existingConfig.GURU_USERNAME || 'guru').trim(), 'Username akun guru', now],
    ['GURU_PASSWORD', String(input.GURU_PASSWORD || existingConfig.GURU_PASSWORD || '123456').trim(), 'Password akun guru', now],
    ['GURU_NAMA', String(input.GURU_NAMA || existingConfig.GURU_NAMA || 'Guru').trim(), 'Nama guru', now],
    ['GURU_KELAS', String(input.GURU_KELAS || existingConfig.GURU_KELAS || '').trim(), 'Kelas yang diampu guru', now],
    ['UPDATED_AT', now, 'Waktu terakhir konfigurasi diperbarui', now]
  ];

  var values = sheet.getDataRange().getValues();
  var keyIndex = 0;
  var rowMap = {};
  for (var i = 1; i < values.length; i++) {
    var key = String(values[i][keyIndex] == null ? '' : values[i][keyIndex]).trim();
    if (key) rowMap[key] = i + 1;
  }

  configRows.forEach(function(row) {
    if (rowMap[row[0]]) {
      sheet.getRange(rowMap[row[0]], 1, 1, 4).setValues([row]);
    } else {
      sheet.appendRow(row);
    }
  });

  return { success: true, message: 'Konfigurasi aplikasi berhasil disimpan.', data: getAppConfigFromSheet() };
}

function getAppConfigFromSheet() {
  var ss = getRequestSpreadsheet_();
  var sheet = ss.getSheetByName('Config');
  if (!sheet || sheet.getLastRow() < 2) {
    return {
      SPREADSHEET_ID: ss.getId(),
      APPSCRIPT_URL: '',
      SCRIPT_ID: '',
      DEFAULT_STUDENT_PASSWORD: '123456',
      UPDATED_AT: ''
    };
  }

  var values = sheet.getDataRange().getValues();
  var result = {};
  for (var i = 1; i < values.length; i++) {
    var key = String(values[i][0] == null ? '' : values[i][0]).trim();
    if (key) result[key] = values[i][1];
  }
  if (!result.SPREADSHEET_ID) result.SPREADSHEET_ID = ss.getId();
  if (!result.DEFAULT_STUDENT_PASSWORD) result.DEFAULT_STUDENT_PASSWORD = '123456';
  return result;
}

function getAppConfig(key) {
  var config = getAppConfigFromSheet();
  if (key) return config[key] !== undefined ? config[key] : '';
  return config;
}

function ensureDefaultTeacherAccount() {
  var ss = getRequestSpreadsheet_();
  var sheet = ss.getSheetByName('Users');
  if (!sheet) return;
  var cfg = getAppConfigFromSheet();
  var account = {
    ID: 'USR-GURU-' + String(cfg.GURU_USERNAME || 'guru').toUpperCase(),
    Username: String(cfg.GURU_USERNAME || 'guru').trim(),
    Password: String(cfg.GURU_PASSWORD || '123456').trim(),
    Nama: String(cfg.GURU_NAMA || 'Guru').trim(),
    Role: 'guru',
    NIS: '',
    NIP: '',
    Kelas: String(cfg.GURU_KELAS || '').trim(),
    Status: 'aktif'
  };
  var headers = sheet.getRange(1,1,1,sheet.getLastColumn()).getValues()[0];
  var values = sheet.getDataRange().getValues();
  var u = headers.indexOf('Username');
  var r = headers.indexOf('Role');
  var found = -1;

  for (var i = 1; i < values.length; i++) {
    var existingUsername = String(values[i][u] == null ? '' : values[i][u]).trim().toLowerCase();
    var existingRole = String(values[i][r] == null ? '' : values[i][r]).trim().toLowerCase();
    if (existingUsername === account.Username.toLowerCase() && existingRole === 'guru') {
      found = i + 1;
      break;
    }
  }

  var now = new Date().toISOString();
  var row = headers.map(function(h) {
    if (h === 'CreatedAt' || h === 'UpdatedAt') return now;
    return account[h] !== undefined ? account[h] : '';
  });

  if (found > 0) {
    sheet.getRange(found, 1, 1, row.length).setValues([row]);
  } else {
    sheet.appendRow(row);
  }
}

function getStudentsByClass(kelas) {
  var result=getSheetData('Siswa');
  kelas=String(kelas||'').trim();
  if(!kelas) return result;
  result.data=(result.data||[]).filter(function(x){return String(x.Kelas||'').trim()===kelas;});
  return result;
}

// Membuat / memperbarui akun login siswa berdasarkan data pada sheet Siswa.
// Aturan wajib:
//   NIS 1     -> username siswa1
//   NIS 2     -> username siswa2
//   password  -> 123456
//   role      -> siswa
function syncStudentToUser(student) {
  var ss = getRequestSpreadsheet_();
  var usersSheet = ss.getSheetByName('Users');
  if (!usersSheet || !student) return false;

  var nis = String(student.NIS == null ? '' : student.NIS).trim();
  if (!nis) return false;

  var headers = usersSheet.getRange(1, 1, 1, usersSheet.getLastColumn()).getValues()[0];
  var values = usersSheet.getDataRange().getValues();
  var nisIndex = headers.indexOf('NIS');
  var usernameIndex = headers.indexOf('Username');
  var roleIndex = headers.indexOf('Role');
  if (nisIndex === -1 || usernameIndex === -1 || roleIndex === -1) return false;

  var username = 'siswa' + nis;
  var found = -1;

  for (var i = 1; i < values.length; i++) {
    var rowNis = String(values[i][nisIndex] == null ? '' : values[i][nisIndex]).trim();
    var rowRole = String(values[i][roleIndex] == null ? '' : values[i][roleIndex]).trim().toLowerCase();
    var rowUsername = String(values[i][usernameIndex] == null ? '' : values[i][usernameIndex]).trim().toLowerCase();

    // Cari berdasarkan NIS + role terlebih dahulu. Ini juga memperbaiki akun lama
    // seperti username "siswa" menjadi "siswa1001".
    if ((rowNis === nis && rowRole === 'siswa') || rowUsername === username.toLowerCase()) {
      found = i + 1;
      break;
    }
  }

  var now = new Date().toISOString();
  var account = {
    ID: 'USR-SISWA-' + nis,
    Username: username,
    Password: '123456',
    Nama: student.Nama || '',
    Role: 'siswa',
    NIS: nis,
    NIP: '',
    Kelas: student.Kelas || '',
    Status: (String(student.Status || 'aktif').toLowerCase() === 'aktif') ? 'aktif' : 'nonaktif'
  };

  var row = headers.map(function(h) {
    if (h === 'CreatedAt') {
      if (found > 0) {
        var createdIdx = headers.indexOf('CreatedAt');
        return createdIdx >= 0 ? values[found - 1][createdIdx] : now;
      }
      return now;
    }
    if (h === 'UpdatedAt') return now;
    return account[h] !== undefined ? account[h] : '';
  });

  if (found > 0) {
    usersSheet.getRange(found, 1, 1, headers.length).setValues([row]);
  } else {
    usersSheet.getRange(usersSheet.getLastRow() + 1, 1, 1, headers.length).setValues([row]);
  }
  return true;
}

function syncAllStudentUsers() {
  var students = getSheetData('Siswa').data || [];
  for (var i = 0; i < students.length; i++) {
    syncStudentToUser(students[i]);
  }
}

function saveStudentAndCreateUser(student) {
  var result = addOrUpdateRow('Siswa', student, 'NIS');
  if (!result.success) return result;
  syncStudentToUser(student);
  return {
    success: true,
    message: 'Data siswa berhasil disimpan dan akun login siswa otomatis dibuat/diperbarui.',
    data: { username: 'siswa' + String(student.NIS).trim(), password: '123456', role: 'siswa' }
  };
}

function importStudentsAndCreateUsers(records) {
  var result = importRowsToSheet('Siswa', records, 'NIS');
  if (!result.success) return result;

  // Sinkronkan seluruh siswa agar data lama maupun hasil import memiliki akun.
  syncAllStudentUsers();
  result.message = 'Import siswa selesai dan akun login siswa otomatis disinkronkan.';
  result.data.accountsSynced = (getSheetData('Siswa').data || []).length;
  return result;
}

function deleteStudentAndUser(nis) {
  var result = deleteRowByKey('Siswa', 'NIS', nis);
  if (!result.success) return result;

  var ss = getRequestSpreadsheet_();
  var usersSheet = ss.getSheetByName('Users');
  if (usersSheet && usersSheet.getLastRow() > 1) {
    var values = usersSheet.getDataRange().getValues();
    var headers = values[0];
    var nisIndex = headers.indexOf('NIS');
    var roleIndex = headers.indexOf('Role');
    if (nisIndex >= 0 && roleIndex >= 0) {
      for (var i = values.length - 1; i >= 1; i--) {
        if (String(values[i][nisIndex]).trim() === String(nis).trim() &&
            String(values[i][roleIndex]).trim().toLowerCase() === 'siswa') {
          usersSheet.deleteRow(i + 1);
        }
      }
    }
  }
  return { success: true, message: 'Data siswa dan akun login siswa berhasil dihapus.', data: null };
}

function loginUser(username, password, role) {
  username = String(username == null ? '' : username).trim().toLowerCase();
  password = String(password == null ? '' : password).trim();
  role = String(role == null ? '' : role).trim().toLowerCase();

  if (!username || !password || !role) {
    return { success: false, message: 'Username, Password, dan Role wajib diisi!', data: null };
  }

  if (role !== 'guru' && role !== 'siswa') {
    return { success: false, message: 'Role hanya boleh guru atau siswa!', data: null };
  }

  var users = getSheetData('Users').data || [];
  for (var i = 0; i < users.length; i++) {
    var dbUsername = String(users[i].Username == null ? '' : users[i].Username).trim().toLowerCase();
    var dbPassword = String(users[i].Password == null ? '' : users[i].Password).trim();
    var dbRole = String(users[i].Role == null ? '' : users[i].Role).trim().toLowerCase();

    if (dbUsername === username && dbPassword === password && dbRole === role) {
      if (users[i].Status !== 'aktif') {
        return { success: false, message: 'Akun Anda sedang nonaktif.', data: null };
      }
      return {
        success: true,
        message: 'Login Berhasil',
        data: {
          ID: users[i].ID,
          Username: users[i].Username,
          Nama: users[i].Nama,
          Role: users[i].Role,
          NIS: users[i].NIS,
          NIP: users[i].NIP,
          Kelas: users[i].Kelas
        }
      };
    }
  }
  return { success: false, message: 'Username atau Password salah!', data: null };
}

function getGuruProfile(body) {

function saveGuruProfile(body) {
  body = body || {};
  var username = String(body.username || '').trim().toLowerCase();
  var role = String(body.role || '').trim().toLowerCase();
  var spreadsheetId = String(body.spreadsheetId || body.SPREADSHEET_ID || '').trim();

  var nip = String(body.NIP || '').trim();
  var email = String(body.Email || '').trim();
  var noHp = String(body.NoHP || '').trim();
  var mapelUtama = String(body.MapelUtama || '').trim();
  var kelas = String(body.Kelas || '').trim();
  if (!nama) return { success:false, message:'Nama Guru wajib diisi.', data:null };

    var ss = getSpreadsheetById_(spreadsheetId);

    var guruSheet = ss.getSheetByName('Guru');
    var userValues = usersSheet.getDataRange().getValues();
    var userHeaders = userValues[0];
    var usernameIndex = userHeaders.indexOf('Username');
    var roleIndex = userHeaders.indexOf('Role');
    var foundUserRow = -1;
    var oldNip = '';
    for (var i = 1; i < userValues.length; i++) {
      if (String(userValues[i][usernameIndex] || '').trim().toLowerCase() === username && String(userValues[i][roleIndex] || '').trim().toLowerCase() === 'guru') {
        foundUserRow = i + 1;
        var oldNipIndex = userHeaders.indexOf('NIP');
        oldNip = oldNipIndex >= 0 ? String(userValues[i][oldNipIndex] || '').trim() : '';
        break;
      }
    }
    if (foundUserRow < 0) return { success:false, message:'Akun Guru tidak ditemukan di sheet Users.', data:null };

    var now = new Date().toISOString();
    var updatedUser = userValues[foundUserRow - 1].slice();
    function putUser(field, value) {
      var idx = userHeaders.indexOf(field);
      if (idx >= 0) updatedUser[idx] = value;
    }
    putUser('Nama', nama);
    putUser('NIP', nip);
    putUser('Kelas', kelas);
    putUser('UpdatedAt', now);
    usersSheet.getRange(foundUserRow, 1, 1, userHeaders.length).setValues([updatedUser]);

    var guruValues = guruSheet.getDataRange().getValues();
    var guruHeaders = guruValues[0];
    var guruNipIndex = guruHeaders.indexOf('NIP');
    var guruNameIndex = guruHeaders.indexOf('Nama');

    for (var j = 1; j < guruValues.length; j++) {

      var rowName = guruNameIndex >= 0 ? String(guruValues[j][guruNameIndex] || '').trim().toLowerCase() : '';
      if ((nip && rowNip && rowNip === nip) || (oldNip && rowNip === oldNip) || (!oldNip && rowName === String(body.oldNama || '').trim().toLowerCase())) {
        foundGuruRow = j + 1;
        break;
      }
    }

    var guruRecord = {NIP:nip, Nama:nama, Email:email, NoHP:noHp, MapelUtama:mapelUtama};
    var guruRow = guruHeaders.map(function(h) { return guruRecord[h] !== undefined ? guruRecord[h] : ''; });
    if (foundGuruRow > 0) guruSheet.getRange(foundGuruRow, 1, 1, guruHeaders.length).setValues([guruRow]);
    else guruSheet.appendRow(guruRow);

function isActiveGuru_(body) {
  body = body || {};
  var username = String(body.username || '').trim().toLowerCase();
  var role = String(body.role || '').trim().toLowerCase();
  if (role !== 'guru' || !username) return false;
  var users = getSheetData('Users').data || [];
  return users.some(function(u) {
    return String(u.Username || '').trim().toLowerCase() === username &&
           String(u.Role || '').trim().toLowerCase() === 'guru' &&
           String(u.Status || '').trim().toLowerCase() === 'aktif';
  });
}

function normalizeMenuType_(value) {
  return String(value || 'file').trim().toLowerCase() === 'link' ? 'link' : 'file';
}

function normalizeMenuTargetTab_(value) {
  return String(value || 'same').trim().toLowerCase() === 'blank' ? 'blank' : 'same';
}

function normalizeMenuStatus_(value) {
  return String(value || 'aktif').trim().toLowerCase() === 'nonaktif' ? 'nonaktif' : 'aktif';
}

function validateMenuTarget_(type, target) {
  type = normalizeMenuType_(type);
  target = String(target || '').trim();
  if (!target) return 'Target file/link wajib diisi.';
  if (/^\s*(javascript|data|vbscript):/i.test(target)) return 'Target tidak aman. Gunakan file lokal, http:// atau https://.';
  if (type === 'link' && !/^https?:\/\//i.test(target)) return 'Untuk tipe Link, URL harus diawali http:// atau https://.';
  return '';
}

function getMenuGuru(body) {
  body = body || {};
  var ss = getSpreadsheetById_(String(body.spreadsheetId || body.SPREADSHEET_ID || '').trim());
  ensureDatabaseSheetsForSpreadsheet_(ss);
  var rows = getSheetDataFromSpreadsheet_(ss, 'MenuGuru').map(function(m) {
    return {
      ID: String(m.ID || ''),
      NamaMenu: String(m.NamaMenu || ''),
      Ikon: String(m.Ikon || 'fa-solid fa-link'),
      Tipe: normalizeMenuType_(m.Tipe),
      Target: String(m.Target || ''),
      TargetTab: normalizeMenuTargetTab_(m.TargetTab),
      Urutan: Number(m.Urutan || 0),
      Status: normalizeMenuStatus_(m.Status),
      CreatedAt: m.CreatedAt || '',
      UpdatedAt: m.UpdatedAt || ''
    };
  });
  rows.sort(function(a,b) {
    var ua = Number(a.Urutan || 0), ub = Number(b.Urutan || 0);
    if (ua !== ub) return ua - ub;
    return a.NamaMenu.localeCompare(b.NamaMenu, 'id');
  });
  var includeInactive = String(body.includeInactive || '').toLowerCase() === 'true' || body.includeInactive === true;
  if (!includeInactive) rows = rows.filter(function(m) { return m.Status === 'aktif'; });
  return {success:true, message:'Menu Guru berhasil dimuat.', data:rows};
}

function saveMenuGuru(body) {
  body = body || {};
  if (!isActiveGuru_(body)) return {success:false, message:'Hanya Guru aktif yang dapat mengatur Menu Utama.', data:null};
  var ss = getSpreadsheetById_(String(body.spreadsheetId || body.SPREADSHEET_ID || '').trim());
  ensureDatabaseSheetsForSpreadsheet_(ss);
  var sheet = ss.getSheetByName('MenuGuru');
  var nama = String(body.NamaMenu || '').trim();
  var ikon = String(body.Ikon || 'fa-solid fa-link').trim() || 'fa-solid fa-link';
  var tipe = normalizeMenuType_(body.Tipe);
  var target = String(body.Target || '').trim();
  var targetTab = normalizeMenuTargetTab_(body.TargetTab);
  var status = normalizeMenuStatus_(body.Status);
  var id = String(body.ID || '').trim();

  if (!nama) return {success:false, message:'Nama Menu wajib diisi.', data:null};
  if (nama.length > 50) return {success:false, message:'Nama Menu maksimal 50 karakter.', data:null};
  var targetError = validateMenuTarget_(tipe, target);
  if (targetError) return {success:false, message:targetError, data:null};

  var values = sheet.getDataRange().getValues();
  var headers = values[0];
  var idIndex = headers.indexOf('ID');
  var orderIndex = headers.indexOf('Urutan');
  var createdIndex = headers.indexOf('CreatedAt');
  var foundRow = -1;
  var existingCreatedAt = '';
  for (var i=1; i<values.length; i++) {
    if (id && String(values[i][idIndex] || '').trim() === id) {
      foundRow = i + 1;
      existingCreatedAt = createdIndex >= 0 ? values[i][createdIndex] : '';
      break;
    }
  }
  if (!id) id = 'MNU-' + Date.now() + '-' + Math.floor(Math.random()*1000);

  var urutan = Number(body.Urutan);
  if (!isFinite(urutan) || urutan <= 0) {
    var maxOrder = 0;
    for (var j=1; j<values.length; j++) maxOrder = Math.max(maxOrder, Number(values[j][orderIndex] || 0));
    urutan = maxOrder + 1;
  }

  var now = new Date().toISOString();
  var record = {
    ID:id, NamaMenu:nama, Ikon:ikon, Tipe:tipe, Target:target,
    TargetTab:targetTab, Urutan:urutan, Status:status,
    CreatedAt:existingCreatedAt || now, UpdatedAt:now
  };
  var row = headers.map(function(h){ return record[h] !== undefined ? record[h] : ''; });
  if (foundRow > 0) sheet.getRange(foundRow,1,1,headers.length).setValues([row]);
  else sheet.appendRow(row);
  return {success:true, message:foundRow > 0 ? 'Menu berhasil diperbarui.' : 'Menu berhasil ditambahkan.', data:record};
}

function reorderMenuGuru(body) {
  body = body || {};
  if (!isActiveGuru_(body)) return {success:false, message:'Hanya Guru aktif yang dapat mengatur urutan menu.', data:null};
  var ss = getSpreadsheetById_(String(body.spreadsheetId || body.SPREADSHEET_ID || '').trim());
  ensureDatabaseSheetsForSpreadsheet_(ss);
  var ids = body.ids || body.order || [];
  if (typeof ids === 'string') {
    try { ids = JSON.parse(ids); } catch(e) { ids = ids.split(','); }
  }
  if (!Array.isArray(ids) || !ids.length) return {success:false, message:'Urutan menu tidak valid.', data:null};

  var sheet = ss.getSheetByName('MenuGuru');
  var values = sheet.getDataRange().getValues(), headers = values[0];
  var idIdx = headers.indexOf('ID'), orderIdx = headers.indexOf('Urutan'), updatedIdx = headers.indexOf('UpdatedAt');
  var rowMap = {};
  for (var i=1; i<values.length; i++) rowMap[String(values[i][idIdx] || '').trim()] = i+1;
  var now = new Date().toISOString(), updated = 0;
  ids.forEach(function(id, index) {
    id = String(id || '').trim();
    if (rowMap[id]) {
      sheet.getRange(rowMap[id], orderIdx+1).setValue(index+1);
      if (updatedIdx >= 0) sheet.getRange(rowMap[id], updatedIdx+1).setValue(now);
      updated++;
    }
  });
  return {success:true, message:'Urutan Menu Utama berhasil disimpan.', data:{updated:updated}};
}

function ensureDatabaseSheetsForSpreadsheet_(ss) {
  var schemas = getDatabaseSchemas();
  Object.keys(schemas).forEach(function(sheetName) {
    var headers = schemas[sheetName];
    var sheet = ss.getSheetByName(sheetName);
    if (!sheet) {
      sheet = ss.insertSheet(sheetName);
      sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
      sheet.getRange(1, 1, 1, headers.length).setFontWeight('bold').setBackground('#e2e8f0');
      sheet.setFrozenRows(1);
    }
  });
}

function getSheetDataFromSpreadsheet_(ss, sheetName) {
  var sheet = ss.getSheetByName(sheetName);
  if (!sheet) return [];
  var values = sheet.getDataRange().getValues();
  if (values.length <= 1) return [];
  var headers = values[0];
  return values.slice(1).map(function(row) {
    var obj = {};
    headers.forEach(function(h, i) { obj[h] = row[i]; });
    return obj;
  });
}

function saveGradesBatchAndRebuildPenilaian(body) {
  body = body || {};
  var records = body.records || body.nilai || [];
  if (typeof records === 'string') {
    try { records = JSON.parse(records); } catch (e) { records = []; }
  }
  if (!Array.isArray(records) || !records.length) {
    return {success:false,message:'Tidak ada nilai yang perlu disimpan.',data:{saved:0}};
  }

  var ss = getRequestSpreadsheet_();
  var sheet = ss.getSheetByName('Nilai');
  if (!sheet) {
    return {success:false,message:'Sheet Nilai tidak ditemukan.',data:null};
  }

  var values = sheet.getDataRange().getValues();
  var headers = values[0] || [];
  var idIdx = headers.indexOf('ID');
  if (idIdx < 0) {
    return {success:false,message:'Kolom ID pada sheet Nilai tidak ditemukan.',data:null};
  }

  var rowMap = {};
  for (var i=1; i<values.length; i++) {
    var existingId = String(values[i][idIdx] == null ? '' : values[i][idIdx]).trim();
    if (existingId) rowMap[existingId] = i;
  }

  var nowDate = new Date();
  var updated = 0;
  var inserted = 0;
  var invalid = [];
  var appendRows = [];

  records.forEach(function(record, index) {
    record = record || {};
    var nilai = Number(record.Nilai);

    if (isNaN(nilai) || nilai < 0 || nilai > 100) {
      invalid.push(index + 1);
      return;
    }

    record.ID = String(record.ID || ('NIL-' + Date.now() + '-' + index)).trim();
    if (!record.Tanggal) record.Tanggal = nowDate;

    var rowData = headers.map(function(h) {
      return record[h] !== undefined ? record[h] : '';
    });

    if (rowMap[record.ID] !== undefined) {
      values[rowMap[record.ID]] = rowData;
      updated++;
    } else {
      appendRows.push(rowData);
      inserted++;
    }
  });

  if (values.length > 1) {
    sheet.getRange(2, 1, values.length - 1, headers.length)
      .setValues(values.slice(1));
  }

  if (appendRows.length) {
    sheet.getRange(sheet.getLastRow() + 1, 1, appendRows.length, headers.length)
      .setValues(appendRows);
  }

  var rebuilt = rebuildPenilaianSheet();

  return {
    success: invalid.length === 0,
    message: invalid.length
      ? ('Sebagian nilai gagal disimpan. Baris tidak valid: ' + invalid.join(', ') + '.')
      : ('Sebanyak ' + (updated + inserted) + ' nilai berhasil disimpan sekaligus.'),
    data: {
      saved: updated + inserted,
      updated: updated,
      inserted: inserted,
      invalid: invalid,
      rebuild: rebuilt
    }
  };
}

function addOrUpdateRow(sheetName, dataObj, keyField) {
  var ss = getRequestSpreadsheet_();
  var sheet = ss.getSheetByName(sheetName);
  if (!sheet) return { success: false, message: 'Sheet tidak ditemukan' };
  
  var values = sheet.getDataRange().getValues();
  var headers = values[0];
  var keyVal = dataObj[keyField];
  var rowIndex = -1;
  
  for (var i = 1; i < values.length; i++) {
    var rowKey = values[i][headers.indexOf(keyField)];
    if (rowKey == keyVal) {
      rowIndex = i + 1;
      break;
    }
  }
  
  var rowData = headers.map(function(h) { return dataObj[h] !== undefined ? dataObj[h] : ''; });
  
  if (rowIndex > 0) {
    sheet.getRange(rowIndex, 1, 1, rowData.length).setValues([rowData]);
  } else {
    sheet.appendRow(rowData);
  }
  return { success: true, message: 'Data berhasil disimpan', data: null };
}

// Import Excel: validasi header, normalisasi data, update berdasarkan key, dan batch insert.
function importRowsToSheet(sheetName, records, keyField) {
  var ss = getRequestSpreadsheet_();
  var sheet = ss.getSheetByName(sheetName);
  if (!sheet) return { success: false, message: 'Sheet ' + sheetName + ' tidak ditemukan.', data: null };
  if (!Array.isArray(records) || records.length === 0) {
    return { success: false, message: 'Tidak ada data untuk diimport.', data: null };
  }

  var range = sheet.getDataRange();
  var values = range.getValues();
  var headers = values.length ? values[0].map(String) : [];
  var keyIndex = headers.indexOf(keyField);
  if (keyIndex === -1) return { success: false, message: 'Kolom kunci ' + keyField + ' tidak ditemukan.', data: null };

  var existing = {};
  for (var i = 1; i < values.length; i++) {
    var k = String(values[i][keyIndex] == null ? '' : values[i][keyIndex]).trim();
    if (k) existing[k] = i + 1;
  }

  var inserted = 0, updated = 0, skipped = 0;
  var appendRows = [];
  records.forEach(function(obj) {
    var key = String(obj[keyField] == null ? '' : obj[keyField]).trim();
    if (!key) { skipped++; return; }
    var row = headers.map(function(h) { return obj[h] !== undefined && obj[h] !== null ? obj[h] : ''; });
    if (existing[key]) {
      sheet.getRange(existing[key], 1, 1, headers.length).setValues([row]);
      updated++;
    } else {
      appendRows.push(row);
      existing[key] = -1;
      inserted++;
    }
  });
  if (appendRows.length) {
    sheet.getRange(sheet.getLastRow() + 1, 1, appendRows.length, headers.length).setValues(appendRows);
  }
  return { success: true, message: 'Import ' + sheetName + ' selesai.', data: { inserted: inserted, updated: updated, skipped: skipped, total: records.length } };
}

function deleteRowByKey(sheetName, keyField, keyValue) {
  var ss = getRequestSpreadsheet_();
  var sheet = ss.getSheetByName(sheetName);
  if (!sheet) return { success: false, message: 'Sheet tidak ditemukan' };
  
  var values = sheet.getDataRange().getValues();
  var headers = values[0];
  var colIndex = headers.indexOf(keyField);
  
  for (var i = 1; i < values.length; i++) {
    if (values[i][colIndex] == keyValue) {
      sheet.deleteRow(i + 1);
      return { success: true, message: 'Data berhasil dihapus', data: null };
    }
  }
  return { success: false, message: 'Data tidak ditemukan', data: null };
}

function parseOpenAiText_(json) {
  if (json && json.output_text) return String(json.output_text);
  var out = json && json.output ? json.output : [];
  var parts = [];
  out.forEach(function(item) {
    (item.content || []).forEach(function(c) {
      if (c.text) parts.push(c.text);
    });
  });
  return parts.join('\n').trim();
}

function extractJsonObject_(text) {
  text = String(text || '').trim();
  try { return JSON.parse(text); } catch (e) {}
  var match = text.match(/\{[\s\S]*\}/);
  if (!match) throw new Error('Respons AI bukan JSON yang valid.');
  return JSON.parse(match[0]);
}

    'Nilai harus angka 0 sampai 100. Pahami angka Bahasa Indonesia seperti delapan puluh=80, sembilan puluh lima=95, seratus=100.',
    'Jika pengguna mengucapkan "absen 1" sebagai jenis penilaian, pertahankan sebagai "Absen 1".',
    'Jawaban reply singkat dan natural dalam Bahasa Indonesia.'
  ].join('\n');
  var payload = {
    model: HALO_AI_MODEL,
    input: system + '\n\nKONTEKS DATA:\n' + JSON.stringify(context) + '\n\nPERINTAH GURU:\n' + instruction,
    max_output_tokens: 500
  };
  var response = UrlFetchApp.fetch('https://api.openai.com/v1/responses', {
    method: 'post',
    contentType: 'application/json',
    headers: { Authorization: 'Bearer ' + apiKey },
    payload: JSON.stringify(payload),
    muteHttpExceptions: true
  });
  var code = response.getResponseCode();
  var text = response.getContentText();
  if (code < 200 || code >= 300) throw new Error('OpenAI API error ' + code + ': ' + text.substring(0, 500));
  return extractJsonObject_(parseOpenAiText_(JSON.parse(text)));
}

function haloHalo(body) {
  body = body || {};
  var username = String(body.username || '').trim();
  var role = String(body.role || '').trim().toLowerCase();
  var text = String(body.text || '').trim();
  if (role !== 'guru' || !validateGuruForAI_(username)) return { success: false, message: 'Asisten Halo-Halo hanya dapat digunakan oleh akun Guru aktif.', data: null };
  if (!text) return { success: false, message: 'Perintah kosong.', data: null };
  if (!consumeHaloQuota_(username)) return { success: false, message: 'Kuota Halo-Halo hari ini sudah habis. Batas harian: ' + HALO_AI_DAILY_LIMIT + ' permintaan.', data: null };
  try {
    var ai = callHaloOpenAI_(text, buildHaloContext_());
    if (ai.action === 'search_student' && ai.student && ai.student.NIS) {
      var students = getSheetData('Siswa').data || [];
      var found = students.filter(function(s) { return String(s.NIS || '').trim() === String(ai.student.NIS).trim(); })[0];
      if (found) ai.student = { NIS: found.NIS, Nama: found.Nama, Kelas: found.Kelas };
    }
    if (ai.action === 'input_grade' && ai.intent) {
      var intent = ai.intent;
      var studentRows = getSheetData('Siswa').data || [];
      var student = studentRows.filter(function(s) { return String(s.NIS || '').trim() === String(intent.NIS || '').trim(); })[0];
      if (!student) { ai.action = 'none'; ai.reply = 'NIS tersebut tidak ditemukan di Data Siswa. Silakan periksa kembali NIS.'; }
      else {
        intent.NIS = String(student.NIS);
        intent.NamaSiswa = student.Nama;
        intent.Kelas = intent.Kelas || student.Kelas;
        var grade = Number(intent.Nilai);
        if (!isFinite(grade) || grade < 0 || grade > 100) { ai.action='none'; ai.reply='Nilai harus berupa angka antara 0 sampai 100.'; }
        else {
          var maps = getSheetData('MasterMapel').data || [];
          if (!intent.KodeMapel && intent.NamaMapel) {
            var foundMapel = maps.filter(function(m) { return String(m.NamaMapel || '').trim().toLowerCase() === String(intent.NamaMapel).trim().toLowerCase() && String(m.Status || 'aktif').toLowerCase() !== 'nonaktif'; })[0];
            if (foundMapel) { intent.KodeMapel = foundMapel.Kode; intent.NamaMapel = foundMapel.NamaMapel; }
          }
          if (!intent.KodeMapel) {
            ai.action = 'none';
            ai.reply = 'Mapel belum disebutkan. Silakan sebutkan nama atau kode mapelnya terlebih dahulu.';
          } else {
            var validMapel = maps.filter(function(m) { return String(m.Kode || '').trim().toLowerCase() === String(intent.KodeMapel).trim().toLowerCase() && String(m.Status || 'aktif').toLowerCase() !== 'nonaktif'; })[0];
            if (!validMapel) { ai.action='none'; ai.reply='Kode atau nama mapel tersebut tidak ditemukan di Master Mapel.'; }
            else { intent.KodeMapel = validMapel.Kode; intent.NamaMapel = validMapel.NamaMapel; intent.Nilai = grade; }
          }
        }
      }
      ai.intent = intent;
    }
    return { success: true, message: 'OK', data: ai };
  } catch (e) {
    return { success: false, message: e.message || String(e), data: null };
  }
}

function haloHaloConfirm(body) {
  body = body || {};
  var username = String(body.username || '').trim();
  var role = String(body.role || '').trim().toLowerCase();
  var intent = body.intent || {};
  if (role !== 'guru' || !validateGuruForAI_(username)) return { success:false, message:'Konfirmasi hanya dapat dilakukan oleh Guru aktif.', data:null };
  var nis = String(intent.NIS || '').trim();
  var kode = String(intent.KodeMapel || '').trim();
  var jenis = String(intent.JenisPenilaian || '').trim();
  var nilai = Number(intent.Nilai);
  if (!nis || !kode || !jenis || !isFinite(nilai) || nilai < 0 || nilai > 100) return { success:false, message:'Data nilai belum lengkap. Pastikan NIS, mapel, jenis penilaian, dan nilai sudah benar.', data:null };
  var students = getSheetData('Siswa').data || [];
  var student = students.filter(function(s){ return String(s.NIS || '').trim() === nis; })[0];
  if (!student) return { success:false, message:'Siswa dengan NIS ' + nis + ' tidak ditemukan.', data:null };
  var subjects = getSheetData('MasterMapel').data || [];
  var subject = subjects.filter(function(m){ return String(m.Kode || '').trim().toLowerCase() === kode.toLowerCase() && String(m.Status || 'aktif').toLowerCase() !== 'nonaktif'; })[0];
  if (!subject) return {success:false, message:'Mapel ' + kode + ' tidak ditemukan atau nonaktif.', data:null};
  var payload = { ID:'NIL-AI-' + Date.now(), Kelas: student.Kelas || intent.Kelas || '', KodeMapel: subject.Kode, NIS: student.NIS, NamaSiswa: student.Nama, JenisPenilaian: jenis, Nilai: nilai, Tanggal: Utilities.formatDate(new Date(), Session.getScriptTimeZone() || 'Asia/Jakarta', 'yyyy-MM-dd') };
  var result = addOrUpdateRow('Nilai', payload, 'ID');
  if (result.success) result.message = 'Nilai berhasil disimpan: ' + student.Nama + ' (' + nis + ') = ' + nilai + ' untuk ' + jenis + ' pada ' + subject.NamaMapel + '.';
  return result;
}

/**
 * Menerima data yang dikumpulkan saat MODE OFFLINE lalu memasukkannya
 * ke Spreadsheet ketika pengguna menekan menu Sinkronisasi.
 * Format body.recordsBySheet:
 * {
 *   Siswa: [...],
 *   MasterMapel: [...],
 *   Absensi: [...],
 *   Nilai: [...],
 *   Jurnal: [...],
 *   Pengumuman: [...]
 * }
 */
function syncOfflineData(body) {
  body = body || {};
  var recordsBySheet = body.recordsBySheet || {};
  var allowed = {
    Siswa: 'NIS',
    MasterMapel: 'Kode',
    Absensi: 'ID',
    Nilai: 'ID',
    Jurnal: 'ID',
    Pengumuman: 'ID',
    HasilKuis: 'ID'
  };
  var summary = {};
  var total = 0;

  Object.keys(recordsBySheet).forEach(function(sheetName) {
    if (!allowed[sheetName]) return;
    var records = recordsBySheet[sheetName];
    if (!Array.isArray(records) || !records.length) return;

    var result = importRowsToSheet(sheetName, records, allowed[sheetName]);
    summary[sheetName] = {
      success: !!result.success,
      count: records.length,
      message: result.message || ''
    };
    if (result.success) total += records.length;
  });

  return {
    success: true,
    message: total
      ? 'Sinkronisasi selesai. ' + total + ' data berhasil dikirim ke Spreadsheet.'
      : 'Tidak ada data offline yang perlu disinkronkan.',
    data: { total: total, sheets: summary }
  };
}

function clearAllData() {
  var ss = getRequestSpreadsheet_();
  var sheets = ss.getSheets();
  for (var i = 0; i < sheets.length; i++) {
    var sheet = sheets[i];
    var lastRow = sheet.getLastRow();
    if (lastRow > 1) {
      sheet.deleteRows(2, lastRow - 1);
    }
  }
  return { success: true, message: 'Seluruh isi data spreadsheet telah dikosongkan (Header tetap dipertahankan).', data: null };
}
